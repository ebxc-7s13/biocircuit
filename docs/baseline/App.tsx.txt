function PostCard({ post, onLike, onSave, onOpen, onToast }: { post: Post; onLike: () => void; onSave: () => void; onOpen: () => void; onToast: (message: string) => void }) {
  const copyPostLink = async () => {
    const shareUrl = new URL(window.location.href)
    shareUrl.hash = `post-${post.id}`
    try { await navigator.clipboard.writeText(shareUrl.toString()); onToast('Post link copied to clipboard') }
    catch { onToast(`Clipboard unavailable · post reference ${post.id}`) }
  }
  const kindLabels: Record<Kind, string> = { Discussion: 'FIELD NOTE', Question: 'ASK THE COMMUNITY', Project: 'PROJECT LOG', Journal: 'PAPER DISCUSSION' }
  return (
    <article className="post-card" id={`post-${post.id}`}>
      <div className="post-author-row"><div className={`avatar avatar-${post.accent}`}>{post.initials}</div><div className="author-text"><div className="author-name-line"><b>{post.author}</b>{post.verified && <span className="tiny-verified" title="Community verified"><ShieldCheck size={13} /></span>}<span className="author-separator">·</span><span className="post-time">{post.time}</span></div><p>{post.role} <span>·</span> {post.institution}</p></div><button className="more-button" aria-label="More post options" onClick={() => onToast('Post options: report, copy link, or mute topic')}><MoreHorizontal size={19} /></button></div>
      <div className="post-kind-row"><span className={`post-kind kind-${post.kind.toLowerCase()}`}><span className="kind-dot" />{kindLabels[post.kind]}</span>{post.kind === 'Question' && post.accepted && <span className="answered-label"><CheckCircle2 size={12} /> ANSWERED</span>}</div>
      <button className="post-title-button" onClick={onOpen}><h2>{post.title}</h2></button>
      <p className="post-body">{post.body}</p>
      {post.kind === 'Journal' && <div className="journal-attachment"><div className="journal-attachment-art"><div className="attachment-orb orb-one"/><div className="attachment-orb orb-two"/><div className="attachment-orb orb-three"/><span>λ / 405</span></div><div className="journal-attachment-copy"><span>{post.journal ?? 'JOURNAL CLUB NOTE'}</span><b>{post.imageLabel ?? 'RESEARCH ARTICLE'}</b><small>{post.citation ?? 'Community reading note · 2026'}</small></div><button className="attachment-open" onClick={onOpen} aria-label="Open journal discussion"><ArrowUpRight size={16}/></button></div>}
      {post.kind === 'Project' && <div className="project-attachment"><div className="project-schematic"><div className="schematic-node node-a"><Microscope size={18}/></div><span className="schematic-link link-a"/><span className="schematic-link link-b"/><div className="schematic-node node-b"><Zap size={16}/></div><div className="schematic-node node-c"><Activity size={16}/></div><span className="schematic-caption">OPTICAL PATH / REV 03</span></div><div className="project-attachment-copy"><span>BUILD LOG · OPEN HARDWARE</span><b>Prototype notes &amp; component choices</b><small>Updated by the project team</small></div><button className="attachment-open" onClick={onOpen} aria-label="Open project"><ArrowUpRight size={16}/></button></div>}
      <div className="post-tags">{post.tags.map((tag) => <button key={tag} onClick={() => onToast(`Filtering by “${tag}” is available through search`)}>{tag}</button>)}</div>
      <div className="post-actions"><div className="post-action-group"><button onClick={onLike} className={post.liked ? 'action-liked' : ''} aria-label={post.liked ? 'Unlike post' : 'Like post'}><Heart size={17} fill={post.liked ?     </article>
  )
}

function Composer({ onClose, onSubmit }: { onClose: () => void; onSubmit: (post: Omit<Post, 'id' | 'author' | 'initials' | 'role' | 'institution' | 'verified' | 'time' | 'likes' | 'replies' | 'saves' | 'accent'>) => void }) {
  const [kind, setKind] = useState<Kind>('Discussion')
  const [title, setTitle] = useState('')
  const [body, setBody] = useState('')
  const [tags, setTags] = useState('')
  const [includeCitation, setIncludeCitation] = useState(false)
  const [error, setError] = useState('')
  const handleSubmit = (event: FormEvent) => {
    event.preventDefault()
    if (title.trim().length < 8 || body.trim().length < 20) { setError('Add a clearer title (8+ characters) and enough context (20+ characters).'); return }
    const parsedTags = tags.split(',').map((tag) => tag.trim()).filter(Boolean).slice(0, 4)
    onSubmit({ kind, title: title.trim(), body: body.trim(), tags: parsedTags.length ? parsedTags : ['Biomedical engineering'], citation: includeCitation ? 'Citation supplied by author · please verify before reuse' : undefined, journal: kind === 'Journal' ? 'COMMUNITY PAPER NOTE' : undefined, imageLabel: kind === 'Journal' ? 'READING NOTE / AUTHOR ADDED' : undefined, accepted: false })
  }
  return <div className="modal-backdrop" onMouseDown={(e) => { if (e.target === e.currentTarget) onClose() }}><form className="modal composer-modal" onSubmit={handleSubmit}><div className="modal-header"><div><span className="eyebrow">CONTRIBUTE TO THE CIRCUIT</span><h2>Put an idea on the table<span className="heading-period">.</span></h2></div><button type="button" className="icon-button" aria-label="Close" onClick={onClose}><X size={19}/></button></div><div className="composer-kind-grid">{([{value:'Discussion',label:'Field note',desc:'Share a learning',icon:FileText},{value:'Question',label:'Ask',desc:'Get expert input',icon:CircleHelp},{value:'Project',label:'Project log',desc:'Show your build',icon:FlaskConical},{value:'Journal',label:'Paper note',desc:'Discuss a paper',icon:BookOpen}] as const).map(({value,label,desc,icon:Icon}) => <button type="button" key={value} className={kind === value ? 'composer-kind active' : 'composer-kind'} onClick={() => setKind(value)}><Icon size={17}/><b>{label}</b><small>{desc}</small></button>)}</div><label className="field-label" htmlFor="post-title">Title <span>Be specific. Cite sources. Share limitations.</span><button className="primary-button" type="submit">Publish post <ArrowRight size={15}/></button></div></form></div>
}
